01Fast I/O bufio — never raw fmt.Scan in loopsI/O

package main
import ("bufio"; "fmt"; "os"; "strconv"; "strings")

func main() {
    in := bufio.NewReader(os.Stdin)           // buffered stdin ★
    out := bufio.NewWriter(os.Stdout)         // buffered stdout ★
    defer out.Flush()                         // or nothing is printed!
    var n int
    fmt.Fscan(in, &n)                         // whitespace-separated tokens
    a := make([]int, n)
    for i := range a { fmt.Fscan(in, &a[i]) }
    fmt.Fprintln(out, n, a[0])
}
// faster token reader for 10^6+ ints: bufio.Scanner + ScanWords
sc := bufio.NewScanner(os.Stdin)
sc.Buffer(make([]byte, 1<<20), 1<<20)         // raise 64KB line limit ★
sc.Split(bufio.ScanWords)
next := func() int { sc.Scan(); v, _ := strconv.Atoi(sc.Text()); return v }
n := next()

16Producer → Multiple Consumers worker pool ★CONCURRENCY

func worker(id int, jobs <-chan int, results chan<- Result, wg *sync.WaitGroup) {
    defer wg.Done()
    for j := range jobs {                    // each job goes to exactly one worker
        results <- Result{id, j, j * j}       // channels are MPMC-safe by design ★
    }
}
func main() {
    jobs := make(chan int, 100)
    results := make(chan Result, 100)
    var wg sync.WaitGroup
    for w := 1; w <= 4; w++ {                 // N consumers
        wg.Add(1)
        go worker(w, jobs, results, &wg)
    }
    go func() {                               // 1 producer
        for i := 0; i < 20; i++ { jobs <- i }
        close(jobs)                           // signals workers: no more work ★
    }()
    go func() { wg.Wait(); close(results) }() // close results after workers exit ★
    for r := range results { fmt.Println(r) } // consume until closed
}
// variants
sem := make(chan struct{}, 4)                 // semaphore: limit concurrency to 4
for _, j := range jobs {
    sem <- struct{}{}; go func() { defer func() { <-sem }(); do(j) }() }
// bounded fan-in with errgroup (golang.org/x/sync):
g, ctx := errgroup.WithContext(ctx); g.SetLimit(4)
for _, j := range jobs { g.Go(func() error { return do(ctx, j) }) }
if err := g.Wait(); err != nil { }            // first error cancels ctx
// broadcast to ALL consumers (not a queue): close(done) wakes every <-done ★

Order ① start workers ② producer sends + closes jobs ③ wg.Wait then close results ④ range results. Closing results from a worker = panic when another worker sends.
