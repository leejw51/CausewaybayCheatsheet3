01Fast I/O read all stdin at onceI/O

use std::io::{self, Read, Write, BufRead};
fn main() {
    let mut s = String::new();
    io::stdin().read_to_string(&mut s).unwrap();
    let mut it = s.split_ascii_whitespace().map(|x| x.parse::<i64>().unwrap());
    let n = it.next().unwrap() as usize;
    let v: Vec<i64> = it.by_ref().take(n).collect();
    // buffered output — much faster than println! inside loops ★
    let out = io::stdout(); let mut w = io::BufWriter::new(out.lock());
    writeln!(w, "{} {}", n, v[0]).unwrap();
}
